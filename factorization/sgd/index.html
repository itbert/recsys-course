SGD для наблюдаемых рейтингов

Производные одной строки​

Пусть e=rui−r^uie=r_{ui}-\widehat r_{ui}e=rui​−rui​. Мгновенный градиент половины квадрата равен −e-e−e по обоим смещениям, −eqi-e q_i−eqi​ по pup_upu​ и −epu-e p_u−epu​ по qiq_iqi​. Для шага величины η\etaη записываем обновление в направлении уменьшения:

pu+=pu+η(eqi−λupu),qi+=qi+η(epu−λiqi).p_u^{+}=p_u+\eta(e q_i-\lambda_u p_u),\qquad q_i^{+}=q_i+\eta(e p_u-\lambda_i q_i).pu+​=pu​+η(eqi​−λu​pu​),qi+​=qi​+η(epu​−λi​qi​).

По аналогии bu+=bu+η(e−λubu)b_u^+=b_u+\eta(e-\lambda_u b_u)bu+​=bu​+η(e−λu​bu​) и ci+=ci+η(e−λici)c_i^+=c_i+\eta(e-\lambda_i c_i)ci+​=ci​+η(e−λi​ci​). В обеих формулах используются старые векторы. Если вычислить вторую производную после обновления первой стороны, получится другой шаг. Малое η\etaη уменьшит разницу, но проверка конечными разностями должна сравнивать производные в одной точке.

На какую цель приходится штраф​

Пусть dud_udu​ — число строк пользователя, did_idi​ — число строк объекта. В учебном коде для строки берём λu=λ/du\lambda_u=\lambda/d_uλu​=λ/du​, λi=λ/di\lambda_i=\lambda/d_iλi​=λ/di​.

Лемма · Сумма локальных штрафов#

При фиксированных параметрах сумма локальных целей по всем строкам равна JJJ из предыдущей главы, если пользовательская и объектная части штрафа делятся соответственно на dud_udu​ и did_idi​.

Доказательство

Блок пользователя входит ровно в dud_udu​ локальных целей. Сумма его коэффициентов равна du(λ/du)=λd_u(\lambda/d_u)=\lambdadu​(λ/du​)=λ. То же для объекта. Нулевые по степени блоки в нашем соглашении фиксированы в нуле и не добавляют штрафа. Для SVD++ аналогичный подсчёт выполняется для каждого вектора истории.

□

При равномерной независимой выборке строки ожидаемый локальный градиент равен ∇J/∣Ω∣\nabla J/|\Omega|∇J/∣Ω∣. Код проходит перемешанную перестановку без повторов внутри эпохи. Градиенты считаются при меняющихся параметрах, поэтому суммарный эффект эпохи отличается от одного полного градиентного шага. Если использовать одинаковый λ\lambdaλ на каждой наблюдавшейся строке, пользователь с большим dud_udu​ получит больший суммарный штраф.

Реализация​

explicit.py
"""MF по наблюдаемым оценкам. NumPy решает линейные системы и хранит факторы."""
from collections import Counter
from dataclasses import dataclass
import math

import numpy as np


def observation_lists(rows, users, items):
    by_user = [[] for _ in range(users)]
    by_item = [[] for _ in range(items)]
    for u, i, r in rows:
        if not (0 <= u < users and 0 <= i < items and math.isfinite(r)):
            raise ValueError("Некорректное наблюдение")
        by_user[u].append((i, r))
        by_item[i].append((u, r))
    return by_user, by_item


@dataclass
class Model:
    mu: float
    b: np.ndarray
    c: np.ndarray
    p: np.ndarray
    q: np.ndarray
    y: np.ndarray
    history: list

    def user_vector(self, user):
        h = self.history[user]
        return self.p[user] + self.y[h].sum(axis=0) / math.sqrt(len(h)) if len(h) else self.p[user]

    def predict(self, user, item):
        return self.mu + self.b[user] + self.c[item] + self.user_vector(user) @ self.q[item]


def rating_objective(rows, model, regularization):
    errors = [r - model.predict(u, i) for u, i, r in rows]
    penalty = sum(np.square(a).sum() for a in (model.b, model.c, model.p, model.q, model.y))
    return .5 * (np.dot(errors, errors) + regularization * penalty)


def initialize(rows, users, items, factors, rng, histories):
    if not rows or factors < 1:
        raise ValueError("Нужны наблюдения и хотя бы один фактор")
    by_user, by_item = observation_lists(rows, users, items)
    p = rng.normal(0, .1, (users, factors))
    q = rng.normal(0, .1, (items, factors))
    p[[not row for row in by_user]] = 0
    q[[not row for row in by_item]] = 0
    h = [np.array(sorted(set(histories.get(u, ()))), dtype=int) for u in range(users)]
    if any(np.any(ids < 0) or np.any(ids >= items) for ids in h):
        raise ValueError("История содержит ID вне каталога")
    mu = math.fsum(r for _, _, r in rows) / len(rows)
    model = Model(mu, np.zeros(users), np.zeros(items), p, q, np.zeros_like(q), h)
    return model, by_user, by_item


def rating_sgd_step(model, u, i, rating, step, reg_u, reg_i, reg_history):
    p_old, q_old = model.p[u].copy(), model.q[i].copy()
    h = model.history[u]
    vector = p_old if not len(h) else model.user_vector(u)
    error = rating - (model.mu + model.b[u] + model.c[i] + vector @ q_old)
    model.b[u] += step * (error - reg_u * model.b[u])
    model.c[i] += step * (error - reg_i * model.c[i])
    model.p[u] += step * (error * q_old - reg_u * p_old)
    model.q[i] += step * (error * vector - reg_i * q_old)
    if len(h):
        model.y[h] += step * (error * q_old / math.sqrt(len(h)) -
                              reg_history[:, None] * model.y[h])


def fit_sgd(rows, users, items, factors=4, epochs=35, step=.025, regularization=.6,
            seed=9, histories=None):
    if epochs < 1 or step <= 0 or regularization < 0:
        raise ValueError("Проверьте параметры SGD")
    rng = np.random.default_rng(seed)
    model, by_user, by_item = initialize(rows, users, items, factors, rng, histories or {})
    history_occurrences = Counter(i for u, _, _ in rows for i in model.history[u])
    history_reg = [np.array([regularization / history_occurrences[i] for i in ids])
                   if by_user[u] else np.zeros(len(ids))
                   for u, ids in enumerate(model.history)]
    losses = [rating_objective(rows, model, regularization)]
    for _ in range(epochs):
        for j in rng.permutation(len(rows)):
            u, i, r = rows[j]
            rating_sgd_step(model, u, i, r, step,
                            regularization / len(by_user[u]),
                            regularization / len(by_item[i]), history_reg[u])
        losses.append(rating_objective(rows, model, regularization))
    return model, losses


def ridge_block(design, target, regularization):
    normal = design.T @ design + regularization * np.eye(design.shape[1])
    return np.linalg.solve(normal, design.T @ target)


def update_blocks(rows, mu, other_bias, other_factors, bias, factors, regularization):
    for j, observations in enumerate(rows):
        if not observations:
            bias[j], factors[j] = 0, 0
            continue
        ids, ratings = zip(*observations)
        ids = list(ids)
        design = np.column_stack((np.ones(len(ids)), other_factors[ids]))
        answer = ridge_block(design, np.asarray(ratings) - mu - other_bias[ids], regularization)
        bias[j], factors[j] = answer[0], answer[1:]


def fit_als(rows, users, items, factors=4, sweeps=18, regularization=.6, seed=9):
    if sweeps < 1 or regularization <= 0:
        raise ValueError("Нужен положительный штраф и хотя бы один проход")
    rng = np.random.default_rng(seed)
    model, by_user, by_item = initialize(rows, users, items, factors, rng, {})
    losses = [rating_objective(rows, model, regularization)]
    for _ in range(sweeps):
        update_blocks(by_user, model.mu, model.c, model.q, model.b, model.p, regularization)
        update_blocks(by_item, model.mu, model.b, model.p, model.c, model.q, regularization)
        losses.append(rating_objective(rows, model, regularization))
    return model, losses

initialize строит списки по пользователям и объектам и фиксирует μ\muμ. rating_sgd_step копирует текущие факторы, считает ошибку и обновляет только используемые блоки. Без истории стоимость обучающего прохода O(∣Ω∣f)O(|\Omega|f)O(∣Ω∣f), память факторов O((U+I)f)O((U+I)f)O((U+I)f). Функция сохраняет полную обучающую цель в конце каждой эпохи. ALS в этом же модуле разбирается далее, а дополнительный вектор истории — в главе SVD++.

Шаг, число эпох, размерность и штраф проверяют на более раннем временном отрезке. Слишком малый шаг может оставить модель в плохом приближении, слишком большой — дать расходящиеся прогнозы. Гарантии выпуклой оптимизации на полную билинейную модель не переносятся. При скрытых факторах знак улучшения на train и test может отличаться.
