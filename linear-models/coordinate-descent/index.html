Координатный спуск для SLIM

Минимизация по одной координате​

В регрессии для одного объекта заморозим все веса кроме wiw_iwi​. Обозначим текущий остаток через r=y−Zwr=y-Zwr=y−Zw, а столбец признака через zi=Z:iz_i=Z_{:i}zi​=Z:i​. Почему хранится остаток​

Если старое значение равно wiw_iwi​, а новое равно s⋆s^\stars⋆, то новая реконструкция увеличилась на (s⋆−wi)zi(s^\star-w_i)z_i(s⋆−wi​)zi​. Остаток уменьшается на тот же вектор. Вычисление скалярного произведения и изменение остатка требуют обхода только ненулевых строк ziz_izi​. CSC хранит их подряд. Слагаемое ∥zi∥22wi\|z_i\|_2^2 w_i∥zi​∥22​wi​ восстанавливает вклад старого веса в ccc, используя сохранённую норму столбца.

Полная реализация​

slim.py
"""Координатный спуск для SLIM на матрице событий в формате CSC."""

import numpy as np
from scipy.sparse import csc_matrix


def kkt_error(gradient, weights, l1, nonnegative):
    """Максимальная невязка покоординатных условий оптимальности."""
    if nonnegative:
        error = np.where(
            weights > 0, np.abs(gradient + l1), np.maximum(0, -gradient - l1)
        )
    else:
        error = np.where(
            weights != 0, np.abs(gradient + l1 * np.sign(weights)),
            np.maximum(0, np.abs(gradient) - l1),
        )
    return float(error.max(initial=0))


def fit_slim(x, l1, l2, *, nonnegative=True, tol=1e-7, max_sweeps=200):
    """Вернуть разреженную B и диагностику по целевым столбцам."""
    x = csc_matrix(x, dtype=float, copy=True)
    x.sum_duplicates()
    x.eliminate_zeros()
    if not (np.isfinite(l1) and l1 >= 0 and np.isfinite(l2) and l2 > 0):
        raise ValueError("expected finite l1 >= 0 and l2 > 0")
    if not np.isfinite(x.data).all() or not (tol > 0 and max_sweeps >= 1):
        raise ValueError("invalid data or stopping rule")

    columns = [
        (x.indices[start:stop], x.data[start:stop])
        for start, stop in zip(x.indptr[:-1], x.indptr[1:])
    ]
    norm2 = np.array([values @ values for _, values in columns])
    data, indices, indptr = [], [], [0]
    diagnostics = []
    for target, (rows, values) in enumerate(columns):
        residual = np.zeros(x.shape[0])
        residual[rows] = values
        weights = np.zeros(x.shape[1])
        free = np.arange(x.shape[1]) != target
        for sweep in range(1, max_sweeps + 1):
            for feature, (active_rows, feature_values) in enumerate(columns):
                if feature == target or norm2[feature] == 0:
                    continue
                before = weights[feature]
                z = feature_values @ residual[active_rows] + norm2[feature] * before
                shrunk = np.maximum(z - l1, 0) if nonnegative else (
                    np.sign(z) * np.maximum(abs(z) - l1, 0)
                )
                after = shrunk / (norm2[feature] + l2)
                weights[feature] = after
                residual[active_rows] -= (after - before) * feature_values

            gradient = -(x.T @ residual) + l2 * weights
            error = kkt_error(gradient[free], weights[free], l1, nonnegative)
            if error <= tol:
                break

        nonzero = np.flatnonzero(weights)
        data.extend(weights[nonzero])
        indices.extend(nonzero)
        indptr.append(len(data))
        diagnostics.append((sweep, error))
    return csc_matrix((data, indices, indptr), shape=(x.shape[1], x.shape[1])), diagnostics

fit_slim копирует вход в CSC, суммирует дубли и удаляет явно записанные нули. columns хранит индексы строк и значения столбцов. На каждый целевой столбец выделяются плотные временные векторы residual и weights. Целевой индекс пропускается, поэтому диагональ остаётся нулевой. Полностью нулевой признак также пропускается: при λ2>0\lambda_2>0λ2​>0 оптимальное значение его веса равно нулю.

После прохода по всем входам kkt_error вычисляет наибольшую покомпонентную невязку. Для активной координаты это модуль ошибки стационарности, для нулевой — величина нарушения неравенства. При достижении max_sweeps возвращается приближение и фактическая невязка. Сравнивайте её с tol, особенно если несколько столбцов близки к линейной зависимости. Каждая координатная минимизация не повышает цель в точной арифметике; один такой шаг не обеспечивает условия оптимальности остальных координат.
